\hypertarget{md__r_e_a_d_m_e_autotoc_md4}{}\doxysection{\texorpdfstring{🚀 Características Principales}{🚀 Características Principales}}\label{md__r_e_a_d_m_e_autotoc_md4}

\begin{DoxyItemize}
\item {\bfseries{Control Térmico y de Humedad con Histéresis:}}
\begin{DoxyItemize}
\item Banda muerta de temperatura: \$\textbackslash{}pm 0.\+5\textbackslash{},\texorpdfstring{$^\wedge$}{\string^}\textbackslash{}circ\textbackslash{}text\{C\}\$
\item Banda muerta de humedad: \$\textbackslash{}pm 2.\+0\textbackslash{},\%\$
\end{DoxyItemize}
\item {\bfseries{Seguridad y Protección Activa de Hardware:}}
\begin{DoxyItemize}
\item Monitoreo continuo de la temperatura del disipador de la celda Peltier.
\item Corte de emergencia automático si \$\+T\_\{\textbackslash{}text\{disipador\}\} \texorpdfstring{$>$}{>} 50.\+0\textbackslash{},\texorpdfstring{$^\wedge$}{\string^}\textbackslash{}circ\textbackslash{}text\{C\}\$ (apaga la celda inmediatamente y fuerza la ventilación de alivio hasta que la temperatura baje a \$45.0\textbackslash{},\texorpdfstring{$^\wedge$}{\string^}\textbackslash{}circ\textbackslash{}text\{C\}\$).
\end{DoxyItemize}
\item {\bfseries{Multitarea con Free\+RTOS (ESP32-\/\+S3):}}
\begin{DoxyItemize}
\item {\ttfamily sensors\+\_\+task}: Muestreo de sensores DHT11 y ADC calibrados (e\+Fuse + oversampling x16 para LM35).
\item {\ttfamily control\+\_\+task}: Máquina de estados de climatización y control de relés ({\itshape Active LOW}).
\item {\ttfamily display\+\_\+task}: Formateo y transmisión periódica de tramas UART (1 Hz).
\item {\ttfamily cli\+\_\+task}: Consola interactiva por puerto serial para ajuste dinámico de consignas.
\end{DoxyItemize}
\item {\bfseries{Tablero Gráfico en Tiempo Real (RP2040 + TFT ILI9341):}}
\begin{DoxyItemize}
\item Matriz de 3x3 tarjetas de alto contraste con refresco diferencial (sin parpadeo).
\item 9 variables en vivo con animaciones e iconos dinámicos según magnitudes.
\item Modo demostración ({\itshape Demo Mode}) automático en ausencia de señal UART.
\end{DoxyItemize}
\end{DoxyItemize}
